cmake_minimum_required(VERSION 3.16)
project(codex_auto_ping VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)

add_library(codex-auto-ping SHARED src/plugin.cpp)
target_link_libraries(codex-auto-ping PRIVATE Threads::Threads)
target_compile_definitions(codex-auto-ping PRIVATE PLUGIN_VERSION="${PROJECT_VERSION}")
set_target_properties(codex-auto-ping PROPERTIES PREFIX "" OUTPUT_NAME "codex-auto-ping")

if(MSVC)
  target_compile_options(codex-auto-ping PRIVATE /W4 /EHsc)
else()
  target_compile_options(codex-auto-ping PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Do not require libstdc++ to be installed in the minimal CPA runtime image.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(codex-auto-ping PRIVATE -static-libstdc++ -static-libgcc)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(plugin-tests tests/plugin_test.cpp)
  target_link_libraries(plugin-tests PRIVATE Threads::Threads)
  target_compile_definitions(plugin-tests PRIVATE PLUGIN_VERSION="${PROJECT_VERSION}")
  if(MSVC)
    target_compile_options(plugin-tests PRIVATE /W4 /EHsc /UNDEBUG)
  else()
    target_compile_options(plugin-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  endif()
  add_test(NAME plugin-tests COMMAND plugin-tests)
  if(UNIX)
    add_executable(plugin-load-check tests/plugin_load_test.cpp)
    target_link_libraries(plugin-load-check PRIVATE ${CMAKE_DL_LIBS})
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_link_options(plugin-load-check PRIVATE -static-libstdc++ -static-libgcc)
    endif()
    add_test(NAME plugin-load-check
      COMMAND plugin-load-check $<TARGET_FILE:codex-auto-ping>)
  endif()
endif()
